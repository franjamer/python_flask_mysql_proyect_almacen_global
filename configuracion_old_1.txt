# Importa y registra el blueprint de estilos si lo necesitas
from .configuracion_estilos import estilos_bp
from flask import Blueprint, render_template, request, session, redirect, url_for, flash
from .configuracion_utils import VISTAS_CONFIG, inicializar_config
from .configuracion_carga import cargar_configuracion, guardar_configuracion, listar_archivos_config
from .vistas_config_loader import cargar_vistas_config

configuracion_bp = Blueprint('configuracion_bp', __name__)

VISTAS_CONFIG = cargar_vistas_config()

@configuracion_bp.route('/configuracion', methods=['GET', 'POST'])
def configuracion():
    if session.get('perfil') != 'Admin':
        flash('Acceso denegado: solo el perfil Admin puede acceder a configuración.', 'error')
        return redirect(url_for('home_bp.menu'))

    config = cargar_configuracion()
    config = inicializar_config(config, VISTAS_CONFIG)
    
    vista = request.args.get('vista', 'inventario')
    campos = VISTAS_CONFIG.get(vista, [])
    archivos_config = listar_archivos_config()

    if request.method == 'POST':
        # Actualiza los textos del menú si están en el formulario
        for campo in ['central', 'titulo_principal', 'titulo_secundario']:
            nuevo_valor = request.form.get(f'texto_menu_{campo}')
            if nuevo_valor:
                config['textos_menu'][campo] = nuevo_valor

        # Actualiza nombres de columnas si están en el formulario
        for campo in config['nombres_columnas']:
            nuevo_valor = request.form.get(f'nombre_columna_{campo}')
            if nuevo_valor:
                config['nombres_columnas'][campo] = nuevo_valor

        # Actualiza estilos por vista si corresponde
        for vista in config.get('estilos_vistas', {}):
            for clave in ['color_texto_menu', 'color_fondo_menu', 'tamano_texto_menu', 'estilo_texto_menu']:
                nuevo_valor = request.form.get(f'estilo_{clave}_{vista}')
                if nuevo_valor:
                    config['estilos_vistas'][vista][clave] = nuevo_valor

        # Guarda la configuración actualizada
        guardar_configuracion(config)
        flash('Configuración guardada correctamente.', 'success')
        return redirect(url_for('configuracion_bp.configuracion'))

    return render_template(
        'configuracion.html',
        config=config,
        vista=vista,
        campos=campos,
        vistas=VISTAS_CONFIG,
        archivos_config=archivos_config
    )

# Importa y registra el blueprint de estilos si lo necesitas
from .configuracion_estilos import estilos_bp, obtener_estilos_globales, obtener_estilos_locales
from flask import Blueprint, render_template, request, session, redirect, url_for, flash
from .configuracion_utils import VISTAS_CONFIG, inicializar_config
from .configuracion_carga import cargar_configuracion, guardar_configuracion, listar_archivos_config
from .vistas_config_loader import cargar_vistas_config

configuracion_bp = Blueprint('configuracion_bp', __name__)

VISTAS_CONFIG = cargar_vistas_config()

@configuracion_bp.route('/configuracion', methods=['GET', 'POST'])
def configuracion():
    if session.get('perfil') != 'Admin':
        flash('Acceso denegado: solo el perfil Admin puede acceder a configuración.', 'error')
        return redirect(url_for('home_bp.menu'))

    config = cargar_configuracion()
    config = inicializar_config(config, VISTAS_CONFIG)

    # --- GESTIÓN DE ESTILOS ---
    # Determina la subpestaña activa y la vista seleccionada para local
    subpestana = request.args.get('subpestana', 'global')
    vista_local = request.args.get('vista_local') or request.form.get('vista_local') or 'inventario'

    # Valores actuales para Global
    estilos_globales = obtener_estilos_globales(config)
    color_texto_global = estilos_globales['color_texto']
    color_fondo_global = estilos_globales['color_fondo']
    tamano_texto_global = estilos_globales['tamano_texto']
    estilo_texto_global = estilos_globales['estilo_texto']

    # Valores actuales para Local
    estilos_locales = obtener_estilos_locales(config, vista_local)
    color_texto_local = estilos_locales['color_texto']
    color_fondo_local = estilos_locales['color_fondo']
    tamano_texto_local = estilos_locales['tamano_texto']
    estilo_texto_local = estilos_locales['estilo_texto']

    # --- PROCESAMIENTO DE FORMULARIOS ---
    if request.method == 'POST':
        # Cambios en estilos globales
        if 'aplicar_estilo_global' in request.form:
            param = request.form.get('parametro_global', 'todos')
            if param == 'todos' or param == 'color_texto_menu':
                config['estilo_menu']['color_texto_menu'] = request.form.get('global_color_texto_menu', color_texto_global)
            if param == 'todos' or param == 'color_fondo_menu':
                config['estilo_menu']['color_fondo_menu'] = request.form.get('global_color_fondo_menu', color_fondo_global)
            if param == 'todos' or param == 'tamano_texto_menu':
                config['estilo_menu']['tamano_texto_menu'] = request.form.get('global_tamano_texto_menu', tamano_texto_global)
            if param == 'todos' or param == 'estilo_texto_menu':
                config['estilo_menu']['estilo_texto_menu'] = request.form.get('global_estilo_texto_menu', estilo_texto_global)
            guardar_configuracion(config)
            flash('Estilos globales actualizados.', 'success')
            return redirect(url_for('configuracion_bp.configuracion', subpestana='global'))

        # Cambios en estilos locales
        if 'aplicar_estilo_local' in request.form:
            vista_local = request.form.get('vista_local', vista_local)
            if 'estilos_vistas' not in config:
                config['estilos_vistas'] = {}
            if vista_local not in config['estilos_vistas']:
                config['estilos_vistas'][vista_local] = {}
            config['estilos_vistas'][vista_local]['color_texto_menu'] = request.form.get('local_color_texto_menu', color_texto_local)
            config['estilos_vistas'][vista_local]['color_fondo_menu'] = request.form.get('local_color_fondo_menu', color_fondo_local)
            config['estilos_vistas'][vista_local]['tamano_texto_menu'] = request.form.get('local_tamano_texto_menu', tamano_texto_local)
            config['estilos_vistas'][vista_local]['estilo_texto_menu'] = request.form.get('local_estilo_texto_menu', estilo_texto_local)
            guardar_configuracion(config)
            flash(f'Estilos de la vista {vista_local} actualizados.', 'success')
            return redirect(url_for('configuracion_bp.configuracion', subpestana='local', vista_local=vista_local))

    # --- CONTEXTO PARA LA PLANTILLA ---
    return render_template(
        'configuracion_estilo.html',
        subpestana=subpestana,
        # Global
        color_texto=estilos_globales['color_texto'],
        color_fondo=estilos_globales['color_fondo'],
        tamano_texto=estilos_globales['tamano_texto'],
        estilo_texto=estilos_globales['estilo_texto'],
        # Local
        color_texto_local=estilos_locales['color_texto'],
        color_fondo_local=estilos_locales['color_fondo'],
        tamano_texto_local=estilos_locales['tamano_texto'],
        estilo_texto_local=estilos_locales['estilo_texto'],
        vista_seleccionada=vista_local,
        vistas=list(VISTAS_CONFIG.keys()),
        config=config
    )
